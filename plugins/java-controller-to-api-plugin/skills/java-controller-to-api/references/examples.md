## Input (Java: UserController.java)

```java
package com.example.controller;

import com.example.dto.UserQueryDTO;
import com.example.vo.UserVO;
import com.example.common.Result;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import jakarta.validation.constraints.NotNull;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/users")
@Tag(name = "User Management")
public class UserController {
    @Operation(summary = "Get user by ID")
    @GetMapping("/{id}")
    public Result<UserVO> getById(@PathVariable @NotNull Long id) {
        return Result.success(new UserVO());
    }

    @Operation(summary = "Paginated user query")
    @GetMapping
    public Result<PageResult<UserVO>> pageQuery(@Valid UserQueryDTO query) {
        return Result.success(new PageResult<>());
    }

    @Operation(summary = "Create user")
    @PostMapping
    public Result<Long> create(@RequestBody @Valid UserCreateDTO dto) {
        return Result.success(1L);
    }
}
```
